Table Members {
  member_id int [pk, increment]
  first_name varchar(50)
  last_name varchar(50)
  email varchar(100) [unique]
  phone varchar(20)
}

Table Books {
  book_id int [pk, increment]
  title varchar(200)
  category_id int
}

Table Copies {
  copy_id int [pk, increment]
  book_id int
  status varchar(30)
}

Table Authors {
  author_id int [pk, increment]
  name varchar(100)
}

Table Categories {
  category_id int [pk, increment]
  name varchar(100)
}

Table Loans {
  loan_id int [pk, increment]
  member_id int
  copy_id int
  loan_date date
  due_date date
  return_date date
}

Table Reservations {
  reservation_id int [pk, increment]
  member_id int
  book_id int
  reservation_date date
  status varchar(30)
}

Table BookAuthors {
  book_id int
  author_id int

  indexes {
    (book_id, author_id) [pk]
  }
}

Ref: Books.category_id > Categories.category_id
Ref: Copies.book_id > Books.book_id
Ref: Loans.member_id > Members.member_id
Ref: Loans.copy_id > Copies.copy_id
Ref: Reservations.member_id > Members.member_id
Ref: Reservations.book_id > Books.book_id
Ref: BookAuthors.book_id > Books.book_id
Ref: BookAuthors.author_id > Authors.author_id
